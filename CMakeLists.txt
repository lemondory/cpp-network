cmake_minimum_required(VERSION 3.25)
project(CppNetwork VERSION 0.1.0 LANGUAGES CXX)
if(PROJECT_SOURCE_DIR STREQUAL PROJECT_BINARY_DIR)
    message(FATAL_ERROR "Use an out-of-source build.")
endif()
option(CPP_NETWORK_BUILD_TESTS "독립 네트워크 검사" ${PROJECT_IS_TOP_LEVEL})
option(CPP_NETWORK_BUILD_EXAMPLES "실행 예제" ${PROJECT_IS_TOP_LEVEL})
set(CPP_NETWORK_SANITIZER "none" CACHE STRING "none, address, thread")
set_property(CACHE CPP_NETWORK_SANITIZER PROPERTY STRINGS none address thread)
include(cmake/CppNetworkAsio.cmake)

add_library(cpp_network STATIC src/connection.cpp src/establishment.cpp)
add_library(cpp_network::transport ALIAS cpp_network)
set_target_properties(cpp_network PROPERTIES EXPORT_NAME transport CXX_EXTENSIONS OFF)
target_compile_features(cpp_network PUBLIC cxx_std_20)
target_include_directories(cpp_network PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(cpp_network PUBLIC CppNetworkAsio Threads::Threads)
if(WIN32)
    target_link_libraries(cpp_network PUBLIC ws2_32 mswsock)
endif()
if(MSVC)
    target_compile_options(cpp_network PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(cpp_network PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(NOT CPP_NETWORK_SANITIZER STREQUAL "none")
    if(MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        message(FATAL_ERROR "Sanitizers require Clang/GCC.")
    endif()
    if(CPP_NETWORK_SANITIZER STREQUAL "address")
        set(check_flags -fsanitize=address,undefined -fno-omit-frame-pointer)
    elseif(CPP_NETWORK_SANITIZER STREQUAL "thread")
        set(check_flags -fsanitize=thread -fno-omit-frame-pointer)
    else()
        message(FATAL_ERROR "Unknown sanitizer")
    endif()
    target_compile_options(cpp_network PUBLIC ${check_flags})
    target_link_options(cpp_network PUBLIC ${check_flags})
endif()

if(CPP_NETWORK_BUILD_EXAMPLES)
    add_executable(tcp-roundtrip examples/tcp_roundtrip.cpp)
    target_link_libraries(tcp-roundtrip PRIVATE cpp_network::transport)
    add_executable(async-roundtrip examples/async_roundtrip.cpp)
    target_link_libraries(async-roundtrip PRIVATE cpp_network::transport)
endif()
if(CPP_NETWORK_BUILD_TESTS)
    enable_testing()
    add_executable(drain-tests tests/drain_tests.cpp)
    target_link_libraries(drain-tests PRIVATE cpp_network::transport)
    foreach(case cancelled graceful timeout abort lifetime reentrant callback_error write_error concurrent close_race)
        add_test(NAME drain.${case} COMMAND drain-tests ${case})
        set_tests_properties(drain.${case} PROPERTIES TIMEOUT 20)
    endforeach()
    add_executable(establishment-tests tests/establishment_tests.cpp)
    target_link_libraries(establishment-tests PRIVATE cpp_network::transport)
    foreach(case validation exchange refusal timeout cancel pause close lifetime callbacks concurrent)
        add_test(NAME establishment.${case} COMMAND establishment-tests ${case})
        set_tests_properties(establishment.${case} PROPERTIES TIMEOUT 20)
    endforeach()
    add_executable(connection-tests tests/connection_tests.cpp)
    target_link_libraries(connection-tests PRIVATE cpp_network::transport)
    foreach(case bounds roundtrip pause concurrent close_race stalled_peer peer_close callback_error lifetime)
        add_test(NAME connection.${case} COMMAND connection-tests ${case})
        set_tests_properties(connection.${case} PROPERTIES TIMEOUT 20)
    endforeach()
    if(CPP_NETWORK_BUILD_EXAMPLES)
        add_test(NAME connection.example COMMAND tcp-roundtrip)
        set_tests_properties(connection.example PROPERTIES TIMEOUT 20)
        add_test(NAME establishment.example COMMAND async-roundtrip)
        set_tests_properties(establishment.example PROPERTIES TIMEOUT 20)
    endif()
endif()

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS cpp_network EXPORT CppNetworkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION include)
set(package_dir "${CMAKE_INSTALL_LIBDIR}/cmake/CppNetwork")
install(EXPORT CppNetworkTargets NAMESPACE cpp_network:: DESTINATION ${package_dir})
configure_package_config_file(cmake/CppNetworkConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/CppNetworkConfig.cmake" INSTALL_DESTINATION ${package_dir})
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/CppNetworkConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/CppNetworkConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/CppNetworkConfigVersion.cmake" cmake/CppNetworkAsio.cmake
    DESTINATION ${package_dir})
